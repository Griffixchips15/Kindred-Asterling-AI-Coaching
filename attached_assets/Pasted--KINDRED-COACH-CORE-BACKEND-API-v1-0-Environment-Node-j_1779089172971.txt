// KINDRED COACH: CORE BACKEND API v1.0
// Environment: Node.js / Express

const express = require('express');
const cors = require('cors');
const { Pool } = require('pg');

const app = express();
const PORT = process.env.PORT || 5000;

// Middleware
app.use(cors()); // Allows our React frontend to communicate with this API
app.use(express.json()); // Parses incoming JSON data

// Database Connection Pool (Connects to the PostgreSQL schema we built)
// Note: In production, these variables will be stored in a secure .env file
const pool = new Pool({
  user: 'postgres',
  host: 'localhost',
  database: 'kindred_coach_db',
  password: 'secure_password_here',
  port: 5432,
});

// ==========================================
// SECURITY MIDDLEWARE (Authentication Layer)
// ==========================================
// This function acts as the bouncer, ensuring only authorized users can post data.
const authenticateJWT = (req, res, next) => {
    // Placeholder for actual JWT verification logic
    // In production, this validates the user's secure token
    console.log("Authentication check passed.");
    next(); 
};

// ==========================================
// API ROUTES: THE DAILY LOOP
// ==========================================

// 1. Submit Morning Assessment
app.post('/api/morning-assessment', authenticateJWT, async (req, res) => {
    try {
        const { userId, mentalLoadDimension, goals } = req.body;
        
        // Insert the assessment into the database
        const newAssessment = await pool.query(
            `INSERT INTO morning_assessments (user_id, mental_load_dimension) 
             VALUES ($1, $2) RETURNING assessment_id`,
            [userId, mentalLoadDimension]
        );
        
        const assessmentId = newAssessment.rows[0].assessment_id;

        // Insert the daily mini-goals linked to this assessment
        for (const goal of goals) {
            await pool.query(
                `INSERT INTO daily_mini_goals (assessment_id, goal_description) 
                 VALUES ($1, $2)`,
                [assessmentId, goal]
            );
        }

        res.status(201).json({ message: "Morning Baseline securely logged." });
    } catch (err) {
        console.error(err.message);
        res.status(500).json({ error: "Server Error during Morning Assessment." });
    }
});

// 2. Submit Interval Body Scan
app.post('/api/interval-scans', authenticateJWT, async (req, res) => {
    try {
        const { userId, physicalIndications, tertiaryEmotion } = req.body;
        
        await pool.query(
            `INSERT INTO interval_scans (user_id, physical_indications, tertiary_emotion) 
             VALUES ($1, $2, $3)`,
            [userId, physicalIndications, tertiaryEmotion]
        );

        res.status(201).json({ message: "Body Scan securely logged." });
    } catch (err) {
        console.error(err.message);
        res.status(500).json({ error: "Server Error during Body Scan." });
    }
});

// 3. Submit Evening Report & Medication Log
app.post('/api/evening-report', authenticateJWT, async (req, res) => {
    try {
        const { userId, medicationId, medEffectivenessScore, healthyStruggleManagement } = req.body;
        
        // Log Medication Effectiveness
        if (medicationId) {
            await pool.query(
                `INSERT INTO medication_logs (medication_id, status, effectiveness_score) 
                 VALUES ($1, 'Taken', $2)`,
                [medicationId, medEffectivenessScore]
            );
        }

        // Log Evening Report details
        await pool.query(
            `INSERT INTO evening_reports (user_id, struggle_managed_healthily) 
             VALUES ($1, $2)`,
            [userId, healthyStruggleManagement]
        );

        res.status(201).json({ message: "Evening Report securely logged." });
    } catch (err) {
        console.error(err.message);
        res.status(500).json({ error: "Server Error during Evening Report." });
    }
});

// ==========================================
// SERVER INITIALIZATION
// ==========================================
app.listen(PORT, () => {
  console.log(`Kindred Coach secure backend is running on port ${PORT}`);
});
